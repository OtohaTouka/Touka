      <PanelGroup direction="horizontal" id="main-layout">
        {!isIDE && (
          <>
            <Panel 
              ref={leftPanelRef}
              order={1}
              id="left-sidebar-panel"
              defaultSize={isSidebarCollapsed ? 0 : 18} 
              minSize={isSidebarCollapsed ? 0 : 12} 
              collapsible
              onCollapse={() => {
                setIsSidebarCollapsed(true);
                localStorage.setItem("touka_sidebar_collapsed", "true");
              }}
              onExpand={() => {
                setIsSidebarCollapsed(false);
                localStorage.setItem("touka_sidebar_collapsed", "false");
              }}
              className="flex flex-col transition-all duration-300"
            >
              <LeftSidebar
                onChatClick={() => setViewMode("chat")}
                onSoulClick={() => setViewMode("soul")}
                onStudioClick={() => setViewMode("studio")}
                onPulseClick={() => setViewMode("pulse")}
                onVaultClick={() => setViewMode("vault")}
                onSettingsClick={() => setViewMode("settings")}
                activeSessionId={activeSessionId}
                activeMode={viewMode}
                isCollapsedExternal={isSidebarCollapsed}
                onToggleCollapseExternal={(collapsed) => {
                  setIsSidebarCollapsed(collapsed);
                  if (leftPanelRef.current) {
                    if (collapsed) leftPanelRef.current.collapse();
                    else leftPanelRef.current.expand();
                  }
                }}
              />
            </Panel>
            {!isSidebarCollapsed && <PanelResizeHandle className="w-[1px] bg-slate-200 dark:bg-slate-800 hover:bg-primary/50 transition-colors z-30" />}
          </>

          order={2}
          minSize={30}

                {/* Hamburger Menu (Constant position) */}
                {!isIDE && (
                  <button

                        const next = !isSidebarCollapsed;
                        setIsSidebarCollapsed(next);
                        if (leftPanelRef.current) {
                            if (next) leftPanelRef.current.collapse();
                            else leftPanelRef.current.expand();
                        }

                {isIDE ? (
                  <div className="flex items-center overflow-x-auto no-scrollbar h-full w-full">

"bg-white dark:bg-[#1e1e1e] text-primary shadow-[inset_0_2px_0_0_#2b8cee]" : "bg-slate-50 dark:bg-[#2d2d2d] text-slate-500 hover:text-slate-700 dark:hover:text-slate-300")}>

"bg-white dark:bg-slate-700 shadow-sm text-primary" : "text-slate-500 hover:text-slate-700")}><MessageSquare size={16} /> {!isIDE && "聊天大廳"}</button>
                      {sysStatus.mode === 'hybrid' && <button onClick={() => setViewMode("soul")} className={cn("px-3 py-1.5 text-sm font-bold rounded-lg transition-all flex items-center gap-2", viewMode === "soul" ? "bg-white dark:bg-slate-700 shadow-sm text-primary" : "text-slate-500 hover:text-slate-700")}><User size={16} /> {!isIDE && "靈魂核心"}</button>}
                      <button onClick={() => setViewMode("studio")} className={cn("px-3 py-1.5 text-sm font-bold rounded-lg transition-all flex items-center gap-2", viewMode === "studio" ? "bg-white dark:bg-slate-700 shadow-sm text-primary" : "text-slate-500 hover:text-slate-700")}><Monitor size={16} /> {!isIDE && "模型工坊"}</button>
                      {sysStatus.mode === 'hybrid' && (
                        <>
                          <button onClick={() => setViewMode("pulse")} className={cn("px-3 py-1.5 text-sm font-bold rounded-lg transition-all flex items-center gap-2", viewMode === "pulse" ? "bg-white dark:bg-slate-700 shadow-sm text-primary" : "text-slate-500 hover:text-slate-700")}><Activity size={16} /> {!isIDE && "系統脈搏"}</button>
                          <button onClick={() => setViewMode("vault")} className={cn("px-3 py-1.5 text-sm font-bold rounded-lg transition-all flex items-center gap-2", viewMode === "vault" ? "bg-white dark:bg-slate-700 shadow-sm text-primary" : "text-slate-500 hover:text-slate-700")}><Database size={16} /> {!isIDE && "記憶金庫"}</button>
                        </>
                      )}
                    </div>
                  </div>

                {!isIDE && (

                    <BarChart4 size={20} />
                  </button>

              </div>

                  history={history}
                  isIDE={isIDE}
                  isSending={isSending}
                  isWaiting={isWaiting}
                  inputValue={inputValue}
                  setInputValue={setInputValue}
                  attachment={attachment}
                  setAttachment={setAttachment}
                  handleSend={handleSend}

                  handleCopy={handleCopy}
                  handleRegenerate={handleRegenerate}
                  handleDeleteMessage={handleDeleteMessage}
                  handleSwitchMessage={handleSwitchMessage}
                  editingMessageId={editingMessageId}
                  setEditingMessageId={setEditingMessageId}
                  editValue={editValue}
                  setEditValue={setEditValue}
                  handleEditMessage={handleEditMessage}
                  availableModels={availableModels}
                  selectedModel={selectedModel}
                  setSelectedModel={setSelectedModel}
                  scrollAnchorRef={scrollAnchorRef}
                  fileInputRef={fileInputRef}
                  handleFileChange={handleFileChange}

          order={3}

          onExpand={() => setIsRightSidebarOpen(true)}

        >
          <RightSidebar isIDE={isIDE} isOpen={isRightSidebarOpen} onClose={handleToggleRightSidebar} />
